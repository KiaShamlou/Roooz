cmake_minimum_required(VERSION 3.16)
project(rooz_native VERSION 0.1.0 LANGUAGES C)
add_library(rooz_native SHARED rooz_native.c)
target_include_directories(rooz_native PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(rooz_native PRIVATE c_std_99)
target_compile_definitions(rooz_native PRIVATE ROOZ_BUILD ROOZ_SHARED)
if(MSVC)
  target_compile_options(rooz_native PRIVATE /utf-8)
endif()
if(ANDROID)
  target_link_options(rooz_native PRIVATE "-Wl,-z,max-page-size=16384")
endif()
