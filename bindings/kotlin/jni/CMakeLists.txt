cmake_minimum_required(VERSION 3.16)
project(rooz_jni C)
add_library(rooz_jni SHARED rooz_jni.c ../../../core/src/rooz.c)
target_include_directories(rooz_jni PRIVATE ../../../core/include)
target_compile_features(rooz_jni PRIVATE c_std_99)
if(NOT ANDROID)
  find_package(JNI REQUIRED)
  target_include_directories(rooz_jni PRIVATE ${JNI_INCLUDE_DIRS})
endif()
if(MSVC)
  target_compile_options(rooz_jni PRIVATE /utf-8)
endif()

if(ANDROID)
  target_link_options(rooz_jni PRIVATE "-Wl,-z,max-page-size=16384")
endif()
