cmake_minimum_required(VERSION 3.16)
project(rooz VERSION 0.1.0 LANGUAGES C)
option(BUILD_SHARED_LIBS "Build a shared library" ON)
add_library(rooz core/src/rooz.c)
target_include_directories(rooz PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/core/include> $<INSTALL_INTERFACE:include>)
target_compile_features(rooz PUBLIC c_std_99)
if(BUILD_SHARED_LIBS)
  target_compile_definitions(rooz PRIVATE ROOZ_BUILD PUBLIC ROOZ_SHARED)
endif()
if(MSVC)
  target_compile_options(rooz PRIVATE /W4 /utf-8)
else()
  target_compile_options(rooz PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(rooz_tests tests/core_test.c)
  target_link_libraries(rooz_tests PRIVATE rooz)
  if(MSVC)
    target_compile_options(rooz_tests PRIVATE /utf-8)
  endif()
  add_test(NAME core COMMAND rooz_tests)
endif()
install(TARGETS rooz LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(FILES core/include/rooz.h DESTINATION include)
